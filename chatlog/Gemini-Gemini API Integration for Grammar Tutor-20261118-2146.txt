const schema = {
  type: "OBJECT",
  properties: {
    id: { type: "STRING" },
    format: { type: "STRING", enum: ["mcq", "correct", "explain", "transform"] },
    level: { type: "INTEGER" },
    skill: { type: "STRING", enum: ["marker", "form", "meaning"] },
    prompt: { type: "STRING" },
    options: { type: "ARRAY", items: { type: "STRING" } },
    answer: { type: "STRING" },
    acceptable: { type: "ARRAY", items: { type: "STRING" } },
    traps: { type: "OBJECT" },
    modelExplanation: { type: "STRING" },
    feedback: {
      type: "OBJECT",
      properties: {
        correct: { type: "STRING" },
        M1: { type: "STRING" },
        M2: { type: "STRING" },
        other: { type: "STRING" }
      },
      required: ["correct", "M1", "M2", "other"]
    },
    source: { type: "STRING" }
  },
  required: ["id", "format", "level", "skill", "prompt", "answer", "modelExplanation", "feedback", "source"]
};

export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).end();
  const { skill, level, lastTag } = req.body;

const promptText = GENERATE_PROMPT
    .replace('{{SKILL}}', skill)
    .replace('{{LEVEL}}', level)
    .replace('{{LAST_TAG}}', lastTag || 'none');

try {
    const response = await fetch(https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${process.env.GEMINI_API_KEY}, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: promptText }] }],
        generationConfig: {
          responseMimeType: "application/json",
          responseSchema: schema
        }
      })
    });

if (!response.ok) throw new Error('Gemini API Error');
const data = await response.json();
const resultText = data.candidates[0].content.parts[0].text;
const question = JSON.parse(resultText);

// Deterministic marker rule validation
const pLower = question.prompt.toLowerCase();
const aLower = question.answer.toLowerCase();

const pastMarkers = ['yesterday', 'ago', 'last', 'when', 'in 19', 'in 20'];
const perfectMarkers = ['already', 'yet', 'so far', 'ever', 'never', 'since', 'just'];

const hasPast = pastMarkers.some(m => pLower.includes(m));
const hasPerfect = perfectMarkers.some(m => pLower.includes(m));
const isPerfectAnswer = aLower.includes('have') || aLower.includes('has') || aLower.includes("'ve") || aLower.includes("'s");

if (hasPast && isPerfectAnswer) {
  return res.status(502).json({ error: "Validation failed: Past marker requires simple past" });
}
if (hasPerfect && !isPerfectAnswer) {
  return res.status(502).json({ error: "Validation failed: Perfect marker requires present perfect" });
}

res.status(200).json(question);

} catch (error) {
    res.status(502).json({ error: error.message });
  }
}

import { pickFromBank } from './bank.js';

export const AI_GENERATE_PROMPT_REFERENCE = (See api/generate.js for full prompt);
export const AI_EVALUATE_PROMPT_REFERENCE = (See api/evaluate.js for full prompt);

const fetchWithTimeout = async (url, options = {}) => {
  const controller = new AbortController();
  const id = setTimeout(() => controller.abort(), 6000);
  try {
    const res = await fetch(url, { ...options, signal: controller.signal });
    clearTimeout(id);
    return res;
  } catch (err) {
    clearTimeout(id);
    throw err;
  }
};

export async function getNextQuestion({ skill, level, lastTag, askedIds }) {
  try {
    const res = await fetchWithTimeout('/api/generate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ skill, level, lastTag })
    });

if (!res.ok) throw new Error('Generation failed or invalid');
const question = await res.json();
return question;

} catch (err) {
    console.warn("AI generation failed, falling back to bank.", err);
    return { ...pickFromBank({ skill, level, askedIds }), source: "bank" };
  }
}

export async function evaluateExplanation({ question, learnerText }) {
  try {
    const res = await fetchWithTimeout('/api/evaluate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        prompt: question.prompt,
        answer: question.answer,
        modelExplanation: question.modelExplanation,
        learnerText
      })
    });

if (!res.ok) throw new Error('Evaluation failed');
const evaluation = await res.json();
return { ...evaluation, source: "ai" };

} catch (err) {
    console.warn("AI evaluation failed, using keyword fallback.", err);
    return runFallbackEvaluation(learnerText);
  }
}

function runFallbackEvaluation(text) {
  const t = text.toLowerCase();
  if (t.includes('past') && (t.includes('result') || t.includes('now'))) {
    return { verdict: "good", tag: null, hint: "Great job connecting the past to the present.", source: "fallback" };
  } else if (t.includes('for') || t.includes('since')) {
    return { verdict: "partial", tag: "M2", hint: "Check if the action is still true now when using for/since.", source: "fallback" };
  } else if (t.includes('past') || t.includes('yesterday') || t.includes('ago')) {
    return { verdict: "partial", tag: "M1", hint: "Remember that specific finished times require the simple past.", source: "fallback" };
  }
  return { verdict: "poor", tag: null, hint: "Review the rules for finished actions versus visible present results.", source: "fallback" };
}
